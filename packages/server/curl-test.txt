# 2) REGISTER (definido en src/graphql/schema.graphql, AÚN NO cableado en src/index.ts)

  -H "Content-Type: application/json" \
  -d '{"query":"mutation Register($username: String!, $email: String!, $password: String!) { register(username: $username, email: $email, password: $password) { id username email createdAt updatedAt } }","variables":{"username":"testuser","email":"test@example.com","password":"Password123!"}}'
